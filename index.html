git init proyecto-educa
# Commit 1: base → Commit 2: login → Commit 3: registro → Commit 4: estudiantes

git checkout -b login      # rama login
git checkout -b registro   # rama registro
git checkout main && git merge login && git merge registro

git remote add origin https://github.com/daiko777is/Gamificado.git
git push -u origin main
